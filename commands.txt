apiVersion: v1
kind: ConfigMap
metadata:
  name: fluent-bit-config
  namespace: logging
  labels:
    app: fluent-bit
data:
  fluent-bit.conf: |

    [SERVICE]
        Flush                     5
        Log_Level                 info
        Daemon                    off
        Parsers_File              parsers.conf

        HTTP_Server               On
        HTTP_Listen               0.0.0.0
        HTTP_Port                 2020
        storage.metrics           On


    [INPUT]
        Name                      tail
        Tag                       kube.*
        Path                      /var/log/containers/inventory*.log,/var/log/containers/notifactions*.log,/var/log/containers/frontend*.log

        multiline.parser          docker, cri

        DB                        /var/log/flb-storage/flb_kube.db
        DB.locking                true
        Mem_Buf_Limit             50MB
        Skip_Long_Lines           On
        Refresh_Interval          10

    [FILTER]
        Name                      kubernetes
        Match                     kube.*

        Kube_Tag_Prefix           kube.var.log.containers.
        # In-cluster API server (used to look up pod metadata).
        Kube_URL                  https://kubernetes.default.svc:443
        Kube_CA_File              /var/run/secrets/kubernetes.io/serviceaccount/ca.crt
        Kube_Token_File           /var/run/secrets/kubernetes.io/serviceaccount/token
        # Try to parse the app's 'log' field as JSON; on success, merge its
        # keys into the record root (structured logging).
        Merge_Log                 On
   
        K8S-Logging.Parser        On
        K8S-Logging.Exclude       On
        # We only consume labels.app, so fetch pod labels but skip annotations.
        Labels                    On
        Annotations               Off
        # Cache API lookups per pod for 60s to reduce API server load.
        Kube_Meta_Cache_TTL       60s

    # FILTER 2: drop kubelet health/probe noise
    [FILTER]
        Name                      grep
        Match                     kube.*
        Exclude                   log (healthz|readiness|liveness)

    # FILTER 3: strip CRI bookkeeping fields from the record body
    [FILTER]
        Name                      modify
        Match                     kube.*
        Remove                    time
        Remove                    _p
    [FILTER]
        Name          nest
        Match         kube.*
        Operation     lift
        Nested_under  kubernetes
        Add_prefix    k8s_
    # OUTPUT: push to Loki
    [OUTPUT]
        Name                      loki
        Match                     kube.*
        Host                      loki.logging.svc.cluster.local
        Port                      3100

        Labels                    agent=fluent-bit, app=$kubernetes['labels']['app']
        Auto_Kubernetes_Labels    off

        Remove_Keys               kubernetes, stream
        Line_Format               json
        storage.total_limit_size  200M
        Retry_Limit               5
        compress                  gzip

  parsers.conf: |
    # Single-line JSON parser. Only used when a pod explicitly requests it via
    # the annotation  fluentbit.io/parser: docker  (K8S-Logging.Parser must be On).
    #
    # IMPORTANT: we deliberately DO NOT define a '[MULTILINE_PARSER] cri' here.
    # Fluent Bit's built-in 'cri' parser emits the message under 'log' (which
    # Merge_Log needs). Redefining it to emit 'message'/'flag' would break both
    # Merge_Log and leak the flag field into every log line.
    [PARSER]
        Name        docker
        Format      json
        Time_Key    time
        Time_Format %Y-%m-%dT%H:%M:%S.%LZ
        Time_Keep   On
---
